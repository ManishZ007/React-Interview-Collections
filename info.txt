// STEP 1 — Install Node.js
// https://nodejs.org/en/download  (choose LTS version)

// STEP 2 — Install TypeScript compiler globally (one time only)
// npm install -g typescript
// tsc --version   ← verify it works

// STEP 3 — Run any .ts file
// Open terminal inside 1Basics-of-typescript/

// Windows:
// compile.bat 1DataTypes
// compile.bat 2Functions

// Linux / macOS:
// chmod +x compile.sh   ← give permission once
// ./compile.sh 1DataTypes
// ./compile.sh 2Functions

// RULE — pass only the filename, NO .ts extension, NO path prefix.
// The script compiles src/<name>.ts → dist/<name>.js then runs it.
// If compilation fails it stops — it never runs broken code.

// Manual (without the script):
// tsc src/2Functions.ts --ignoreConfig --outDir dist
// node dist/2Functions.js

// ── PHASE 2: React (2Basic-of-ReactTypscript/1myapp/) ───────

// STEP 1 — Install dependencies (one time only)
// cd 2Basic-of-ReactTypscript/1myapp
// npm install

// STEP 2 — Start the dev server
// npm run dev
// Open http://localhost:5173 in your browser.
// The app hot-reloads — save a file and the browser updates instantly.

// STEP 3 — Where to write lessons
// All hook files are inside: src/hooks/
// Context pattern files are inside: src/hooks/Context/
// To test a component, import it into App.tsx and render it.

// Example App.tsx:
// import UseState from "./hooks/UseState"
// function App() { return <UseState /> }

// ── QUICK REFERENCE — Hook decision guide ───────────────────

// useState   → simple value that changes and needs to show in UI
// useEffect  → side effects: fetch, timer, event listener, localStorage
// useReducer → complex state with many fields or many transition types
// useRef     → access DOM element OR store value without causing re-render
// useContext → share data across many components (theme, user, language)

// ── COMMON ERRORS ───────────────────────────────────────────

// "Cannot find module" after adding a new file?
// → Make sure the file extension is .tsx (not .ts) for JSX files.

// TypeScript error on event handler?
// → Type it: (e: React.ChangeEvent<HTMLInputElement>) => void

// useContext returns undefined?
// → You forgot to wrap the component with the Provider in App.tsx or main.tsx

// Hot reload not working?
// → Check the terminal for compile errors. Fix them and the server restarts.

// ── FILE NAMING CONVENTION ───────────────────────────────────

// .tsx  → any file that contains JSX (components, context files)
// .ts   → plain TypeScript with no JSX (utilities, types, hooks that return no JSX)
// Components and hooks start with a capital letter: UseState, ThemeProvider
// Custom hooks start with "use": useTheme, useAuth, useFetch
